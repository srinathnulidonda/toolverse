    { id: 3, text: "Update resume", done: false },
];

export default function TodoWidget() {
    const [tasks, setTasks] = useState<Task[]>(INITIAL);
    const [input, setInput] = useState("");

    const toggle = (id: number) =>
        setTasks(t => t.map(x => x.id === id ? { ...x, done: !x.done } : x));

    const add = () => {
        const text = input.trim();
        if (!text) return;
        setTasks(t => [...t, { id: Date.now(), text, done: false }]);
        setInput("");
    };

    return (
        <div style={{
            background: "var(--bg-card)",
            border: "0.5px solid var(--border)",
            borderRadius: "var(--radius-lg)",
            padding: "14px 16px",
            minWidth: 0,
        }}>
            <div style={{
                fontSize: 11, fontWeight: 600, color: "var(--text-tertiary)",

            }}>
                Today's tasks
            </div>

            <div style={{ display: "flex", flexDirection: "column", gap: 6, marginBottom: 10 }}>
                {tasks.map(task => (
                    <label key={task.id} style={{
                        display: "flex", alignItems: "center", gap: 8,
                        cursor: "pointer",
                    }}>
                        <div onClick={() => toggle(task.id)} style={{
                            width: 14, height: 14,
                            borderRadius: 3,
                            border: task.done ? "none" : "1.5px solid var(--border)",
                            background: task.done ? "var(--brand)" : "transparent",
                            flexShrink: 0, cursor: "pointer",
                            display: "flex", alignItems: "center", justifyContent: "center",
                        }}>
                            {task.done && (
                                <svg width="8" height="8" viewBox="0 0 10 10" fill="none">
                                    <path d="M2 5l2.5 2.5L8 3" stroke="#fff" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
                                </svg>
                            )}
                        </div>
                        <span style={{
                            fontSize: 12,
                            color: task.done ? "var(--text-tertiary)" : "var(--text)",
                            textDecoration: task.done ? "line-through" : "none",

                    </label>

            <div style={{ display: "flex", gap: 6 }}>
                <input

                    onChange={e => setInput(e.target.value)}
                    onKeyDown={e => e.key === "Enter" && add()}
                    placeholder="Add task…"
                    style={{
                        flex: 1, fontSize: 12, padding: "6px 8px",
                        borderRadius: "var(--radius-sm)",
                        border: "0.5px solid var(--border)",
                        background: "var(--bg-surface)",
                        color: "var(--text)",
                        outline: "none",
                    }}
                />
                <button onClick={add} style={{
                    fontSize: 12, padding: "6px 10px",

                    color: "var(--text-secondary)",
                    cursor: "pointer",

                }}>
                    Add
                </button>
